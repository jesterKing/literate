Shared semantic screen rendering

Both screen adapters use accepted publication contributions and the whole-block syntax emitter. Only semantic reference ranges become links; HTML is never rescanned for tags. This module has no filesystem or VS Code dependency.

The adapter chooses destinations without changing logical identity. Fixed-width IDs are shared with LaTeX; URL components, labels and attributes are escaped separately. Preview supplies a synchronous lookup of asynchronously resolved URLs.

export function screenHref(from: string, to: string, target: string, format: ScreenFormat,
    previewDestination?: PreviewDestination): string | undefined {
    if (from === to) { return '#' + target; }
    if (format === 'preview') { return previewDestination?.(target); }
    const base = from.split('/'); base.pop();
    const destination = to.split('/');
    while (base.length && base[0] === destination[0]) { base.shift(); destination.shift(); }
    const relative = [...base.map(() => '..'), ...destination];
    if (format === 'html') { relative[relative.length - 1] = relative[relative.length - 1].replace(/\.literate$/, '.html'); }
    return './' + relative.map(encodeURIComponent).join('/') + '#' + target;
}

Literal ranges and references arrive in source order. A monotonic syntax cursor visits each syntax leaf once (plus split boundaries), rather than rescanning every leaf per reference. A leaf straddling a reference remains available for its suffix.

<<screen syntax merge>>=
function screenSyntax(display: ReturnType<typeof highlightContribution>) {
    let index = 0;
    return (start: number, end: number): string => {
        const output: string[] = [];
        while (index < display.spans.length) {
            const span = display.spans[index];
            if (span.end <= start) { index++; continue; }
            if (span.start >= end) { break; }
            const left = Math.max(start, span.start), right = Math.min(end, span.end);
            const scope = span.scope.split('.')[0];
            if (left < right && scope !== 'comment') {
                const text = escapeHtml(display.text.slice(left, right));
                output.push(scope ? `<span class="hljs-${escapeHtml(scope)}">${text}</span>` : text);
            }
            if (span.end > end) { break; }
            index++;
        }
        return output.join('');
    };
}

const screenBodyTokens = new Set([
    'paragraph_open', 'paragraph_close', 'inline', 'text', 'softbreak', 'hardbreak',
    'bullet_list_open', 'bullet_list_close', 'ordered_list_open', 'ordered_list_close',
    'list_item_open', 'list_item_close', 'em_open', 'em_close', 'strong_open',
    'strong_close', 'code_inline', 'link_open', 'link_close'
]);
function sanitizeAnnotationTokens(tokens: Token[]): Token[] {
    for (const token of tokens) {
        if (!screenBodyTokens.has(token.type)) {
            token.type = 'text'; token.tag = ''; token.nesting = 0;
            token.children = null; token.hidden = false;
        } else if (token.children) { sanitizeAnnotationTokens(token.children); }
        token.attrs = token.attrs?.filter(([name]) =>
            (token.type === 'link_open' && (name === 'href' || name === 'title'))
            || (token.type === 'ordered_list_open' && name === 'start')) || null;
    }
    return tokens;
}

export function installScreenTermLinks(md: MarkdownIt): void {
    const open = md.renderer.rules.link_open, close = md.renderer.rules.link_close;
    const destination = (href: string, context?: ScreenAnnotationContext) => {
        if (!context) { return undefined; }
        const target = resolveTermHref(context.annotations, href);
        const at = target && screenIndex(context.model, context.annotations).targets.get(target)?.definition;
        return target && at ? { target, href: screenHref(context.path, at.path, target,
            context.format, context.destination) } : undefined;
    };
    md.renderer.rules.link_open = (tokens, index, options, env, renderer) => {
        const href = tokens[index].attrGet('href') || '';
        if (!href.startsWith('term:')) { return open ? open(tokens, index, options, env, renderer) : renderer.renderToken(tokens, index, options); }
        const link = destination(href, env?.literateScreen);
        const use = env?.literateUseTargets?.get(tokens[index]) as PublicationTermUse | undefined;
        const anchor = use ? `<span id="${screenTermUseTarget(use)}"${env.literateScreen?.format === 'preview' ? ` class="code-line" data-line="${use.at.line}"` : ''}></span>` : '';
        return anchor + (link?.href ? `<a href="${escapeHtml(link.href)}" data-literate-target="${link.target}">` : '<span class="literate-term-unresolved">');
    };
    md.renderer.rules.link_close = (tokens, index, options, env, renderer) => {
        let start = index - 1;
        while (start >= 0 && tokens[start].type !== 'link_open') { start--; }
        const href = start >= 0 ? tokens[start].attrGet('href') || '' : '';
        if (href.startsWith('term:')) { return destination(href, env?.literateScreen)?.href ? '</a>' : '</span>'; }
        return close ? close(tokens, index, options, env, renderer) : renderer.renderToken(tokens, index, options);
    };
}

export function screenTermUseAnchors(tokens: Token[], annotations: PublicationAnnotations, owner: string) {
    const anchors = new Map<Token, PublicationTermUse>();
    for (const use of annotations.uses) {
        if (use.owner !== owner) { continue; }
        const token = tokens[use.tokenIndex]?.children?.[use.childIndex];
        if (token?.type === 'link_open' && token.attrGet('href')?.startsWith('term:')) { anchors.set(token, use); }
    }
    return anchors;
}

async function connect(url) {
    const socket = new WebSocket(url), pending = new Map();
    let id = 0;
    await new Promise((resolve, reject) => { socket.onopen = resolve; socket.onerror = reject; });
    socket.onmessage = event => {
        const value = JSON.parse(event.data), entry = pending.get(value.id);
        if (entry) {
            pending.delete(value.id); clearTimeout(entry.timer);
            value.error ? entry.reject(value.error) : entry.resolve(value.result);
        }
    };
    return { close: () => socket.close(), send(method, params = {}, sessionId) {
        return new Promise((resolve, reject) => {
            const next = ++id;
            const timer = setTimeout(() => { pending.delete(next); reject(new Error(method + ' timed out')); }, 5000);
            pending.set(next, { resolve, reject, timer });
            socket.send(JSON.stringify({ id: next, method, params, sessionId }));
        });
    } };
}

function observePreviewLink(href, action) {
    const docs = [document, ...Array.from(document.querySelectorAll('iframe')).map(frame => {
        try { return frame.contentDocument; } catch { return null; }
    }).filter(Boolean)];
    for (const doc of docs) {
        const links = Array.from(doc.querySelectorAll('a[href]'));
        const index = links.findIndex(value => value.getAttribute('href') === href);
        if (index < 0) { continue; }
        const link = links[index], target = href.startsWith('#') ? doc.getElementById(href.slice(1)) : null;
        if (action === 'prepare') {
            if (!links[index - 1]) { throw new Error('keyboard fixture needs a preceding anchor'); }
            links[index - 1].focus();
            doc.__m2ClickTrusted = false;
            doc.addEventListener('click', event => { doc.__m2ClickTrusted = event.isTrusted; }, { once: true, capture: true });
        }
        if (action === 'point') { link.scrollIntoView({ block: 'center' }); }
        const rect = link.getBoundingClientRect();
        const style = doc.defaultView.getComputedStyle(link);
        return { x: rect.left + rect.width / 2, y: rect.top + rect.height / 2,
                    focused: doc.activeElement === link, focusVisible: link.matches(':focus-visible'),
            label: link.getAttribute('aria-label'), tabIndex: link.tabIndex, outline: style.outlineStyle,
            outlineWidth: style.outlineWidth, hash: doc.defaultView.location.hash,
            scrollY: doc.defaultView.scrollY, top: target?.getBoundingClientRect().top,
            height: doc.defaultView.innerHeight, line: target?.getAttribute('data-line'),
                        trustedClick: doc.__m2ClickTrusted, ready: doc.readyState === 'complete' };
    }
    return null;
}
